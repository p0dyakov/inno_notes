    // m1 isn’t accessible
    // m2 is accessible
    // m3 is accessible
}

Base b = new B();
b.m1 = 1; // Error
b.m2 = 2; // Error
b.m3 = 3; // OK
    • Public members from the base class are accessible everywhere: either from within the class or from within its derived classes or from outside the class.
    • Protected members of the base class are accessible from within its derived classes only.
    • Private members of the class are accessible only within the class itself.

    (The rules are a bit more complicated in case of multiple inheritance, see later…)

    Inheritance

class A {

    void functionA() {
        // Code for function A
    }

// Base class B
class B {
public:
    void functionB() {
        // Code for function B
    }
};

// Derived class C inheriting from A (publicly) and B (privately)
class C : public A, private B {
public:
    void functionC() {
        // Code for function C
        functionA(); // Call function from class A (public inheritance)
        functionB(); // Call function from class B (private inheritance)
    }
};

int main() {
    C obj;
    obj.functionA(); // Accessible, inherited publicly from class A
    // obj.functionB(); // Not accessible, inherited privately from class B
    obj.functionC(); // Defined in class C, can access functionB internally
    return 0;
}

    Abstract Class

    class Shape {
public:
    // Pure virtual function
    virtual void draw() = 0;
    // Regular method
    void setWidth(int w) {
        width = w;
    }
    void setHeight(int h) {
        height = h;
    }
protected:
    int width;
    int height;
};

// Derived class
class Rectangle : public Shape {
public:
    void draw() override {
        cout << "Drawing a rectangle of width " << width << " and height " << height << endl;
    }
};

int main() {
    // Shape s; // Error: Cannot create an instance of the abstract class 'Shape'
    Rectangle rect;
    rect.setWidth(5);
    rect.setHeight(7);
    rect.draw(); // OK: Rectangle is a concrete class
    return 0;
}

    Polymorphism

    #include <iostream>
using namespace std;

// Base class
class Shape {
public:
    virtual void draw() {
        cout << "Drawing a shape" << endl;
    }
};

// Derived class
class Circle : public Shape {
public:
    void draw() override {
        cout << "Drawing a circle" << endl;
    }
};

// Another derived class
class Rectangle : public Shape {
public:
    void draw() override {
        cout << "Drawing a rectangle" << endl;
    }
};

void drawShape(Shape &shape) {
    // Polymorphic call: the correct draw() is called based
    // on the type of object
    shape.draw();
}

int main() {
    Circle circle;
    Rectangle rectangle;
    drawShape(circle); // Calls Circle's draw()
    drawShape(rectangle); // Calls Rectangle's draw()
    return 0;
}
